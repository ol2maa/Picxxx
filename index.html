// ============ مولّد الصور الفنية العشوائية ============ // قائمة الأنماط الفنية const ART_STYLES = [ 'geometric', // أشكال هندسية 'gradient', // تدرجات لونية 'circles', // دوائر متداخلة 'waves', // موجات 'mosaic', // فسيفساء 'spiral', // حلزوني 'triangles', // مثلثات 'stars', // نجوم ]; // لوحات ألوان جاهزة const COLOR_PALETTES = [ ['#FF6B6B', '#FFE66D', '#4ECDC4', '#1A535C', '#FFE66D'], ['#F38181', '#FCE38A', '#EAFFD0', '#95E1D3', '#AA96DA'], ['#08D9D6', '#252A34', '#FF2E63', '#EAEAEA', '#F9ED69'], ['#A8E6CF', '#DCEDC1', '#FFD3B6', '#FFAAA5', '#FF8B94'], ['#355C7D', '#6C5B7B', '#C06C84', '#F67280', '#F8B195'], ['#2C3E50', '#E74C3C', '#ECF0F1', '#3498DB', '#9B59B6'], ['#FDCB6E', '#E17055', '#D63031', '#FF7675', '#FAB1A0'], ['#00B894', '#00CEC9', '#81ECEC', '#55EFC4', '#DFE6E9'], ['#6D4C41', '#FBC02D', '#F57C00', '#3E2723', '#D7CCC8'], ['#1A1A2E', '#16213E', '#0F3460', '#E94560', '#F2A365'], ]; function randomFrom(arr) { return arr[Math.floor(Math.random() * arr.length)]; } function generateRandomArt() { const size = 800; const canvas = document.createElement('canvas'); canvas.width = size; canvas.height = size; const ctx = canvas.getContext('2d'); const style = randomFrom(ART_STYLES); const palette = randomFrom(COLOR_PALETTES); // رسم حسب النمط switch (style) { case 'geometric': drawGeometric(ctx, size, palette); break; case 'gradient': drawGradient(ctx, size, palette); break; case 'circles': drawCircles(ctx, size, palette); break; case 'waves': drawWaves(ctx, size, palette); break; case 'mosaic': drawMosaic(ctx, size, palette); break; case 'spiral': drawSpiral(ctx, size, palette); break; case 'triangles': drawTriangles(ctx, size, palette); break; case 'stars': drawStars(ctx, size, palette); break; } const img = new Image(); img.src = canvas.toDataURL('image/png'); return img; } // --- دوال رسم الأنماط المختلفة --- function drawGeometric(ctx, size, palette) { ctx.fillStyle = palette[0]; ctx.fillRect(0, 0, size, size); for (let i = 0; i < 25; i++) { ctx.fillStyle = palette[i % palette.length]; ctx.globalAlpha = 0.7 + Math.random() * 0.3; const w = 50 + Math.random() * 200; const h = 50 + Math.random() * 200; const x = Math.random() * (size - w); const y = Math.random() * (size - h); if (Math.random() > 0.5) { ctx.fillRect(x, y, w, h); } else { ctx.beginPath(); ctx.arc(x + w/2, y + h/2, w/2, 0, Math.PI * 2); ctx.fill(); } } ctx.globalAlpha = 1; } function drawGradient(ctx, size, palette) { const dir = Math.random() > 0.5; const grad = dir ? ctx.createLinearGradient(0, 0, size, size) : ctx.createLinearGradient(0, size, size, 0); palette.forEach((c, i) => grad.addColorStop(i / (palette.length - 1), c)); ctx.fillStyle = grad; ctx.fillRect(0, 0, size, size); // إضافة أشكال شفافة for (let i = 0; i < 15; i++) { ctx.fillStyle = palette[i % palette.length]; ctx.globalAlpha = 0.3; ctx.beginPath(); ctx.arc( Math.random() * size, Math.random() * size, 30 + Math.random() * 120, 0, Math.PI * 2 ); ctx.fill(); } ctx.globalAlpha = 1; } function drawCircles(ctx, size, palette) { ctx.fillStyle = palette[0]; ctx.fillRect(0, 0, size, size); const cx = size / 2; const cy = size / 2; const maxR = size / 2; for (let i = 0; i < 40; i++) { const r = (maxR / 40) * i; ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI * 2); ctx.fillStyle = palette[i % palette.length]; ctx.globalAlpha = 0.6; ctx.fill(); } ctx.globalAlpha = 1; } function drawWaves(ctx, size, palette) { ctx.fillStyle = palette[0]; ctx.fillRect(0, 0, size, size); for (let i = 0; i < palette.length * 4; i++) { ctx.beginPath(); ctx.moveTo(0, size); const amplitude = 30 + Math.random() * 60; const frequency = 0.005 + Math.random() * 0.01; const offset = Math.random() * 200; for (let x = 0; x <= size; x += 5) { const y = size / 2 + offset + Math.sin(x * frequency + i) * amplitude; ctx.lineTo(x, y); } ctx.lineTo(size, size); ctx.closePath(); ctx.fillStyle = palette[i % palette.length]; ctx.globalAlpha = 0.5; ctx.fill(); } ctx.globalAlpha = 1; } function drawMosaic(ctx, size, palette) { const gridN = 8; const cell = size / gridN; for (let i = 0; i < gridN; i++) { for (let j = 0; j < gridN; j++) { ctx.fillStyle = palette[Math.floor(Math.random() * palette.length)]; ctx.fillRect(i * cell, j * cell, cell, cell); // شكل داخلي ctx.fillStyle = palette[Math.floor(Math.random() * palette.length)]; ctx.beginPath(); ctx.arc( i * cell + cell/2, j * cell + cell/2, cell/2 * (0.3 + Math.random() * 0.6), 0, Math.PI * 2 ); ctx.fill(); } } } function drawSpiral(ctx, size, palette) { ctx.fillStyle = palette[0]; ctx.fillRect(0, 0, size, size); const cx = size / 2; const cy = size / 2; for (let i = 0; i < 800; i++) { const angle = i * 0.15; const radius = i * 0.5; const x = cx + Math.cos(angle) * radius; const y = cy + Math.sin(angle) * radius; if (x < 0 || x > size || y < 0 || y > size) break; ctx.beginPath(); ctx.arc(x, y, 8 + (i / 800) * 15, 0, Math.PI * 2); ctx.fillStyle = palette[i % palette.length]; ctx.globalAlpha = 0.85; ctx.fill(); } ctx.globalAlpha = 1; } function drawTriangles(ctx, size, palette) { ctx.fillStyle = palette[0]; ctx.fillRect(0, 0, size, size); for (let i = 0; i < 30; i++) { ctx.beginPath(); const x1 = Math.random() * size; const y1 = Math.random() * size; const x2 = Math.random() * size; const y2 = Math.random() * size; const x3 = Math.random() * size; const y3 = Math.random() * size; ctx.moveTo(x1, y1); ctx.lineTo(x2, y2); ctx.lineTo(x3, y3); ctx.closePath(); ctx.fillStyle = palette[i % palette.length]; ctx.globalAlpha = 0.6; ctx.fill(); } ctx.globalAlpha = 1; } function drawStars(ctx, size, palette) { // خلفية داكنة const grad = ctx.createRadialGradient(size/2, size/2, 50, size/2, size/2, size/2); grad.addColorStop(0, palette[2] || '#2C3E50'); grad.addColorStop(1, palette[0]); ctx.fillStyle = grad; ctx.fillRect(0, 0, size, size); // نجوم for (let i = 0; i < 200; i++) { const x = Math.random() * size; const y = Math.random() * size; const r = Math.random() * 4 + 1; ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI * 2); ctx.fillStyle = palette[Math.floor(Math.random() * palette.length)]; ctx.globalAlpha = 0.5 + Math.random() * 0.5; ctx.fill(); } // نجمة كبيرة ctx.globalAlpha = 1; drawStarShape(ctx, size/2, size/2, 100, 5, palette[1]); } function drawStarShape(ctx, cx, cy, r, points, color) { ctx.beginPath(); for (let i = 0; i < points * 2; i++) { const angle = (i * Math.PI) / points - Math.PI / 2; const radius = i % 2 === 0 ? r : r / 2; const x = cx + Math.cos(angle) * radius; const y = cy + Math.sin(angle) * radius; if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y); } ctx.closePath(); ctx.fillStyle = color; ctx.fill(); }
